`; document.getElementById('reviewsContainer').insertAdjacentHTML('beforeend', newReviewHtml); // Reset input fields document.getElementById('reviewerNameInput').value = ''; document.getElementById('reviewTextInput').value = ''; selectStars(5); showToast('شكراً لك! تم إضافة تقييمك بنجاح ⭐'); } function showToast(text) { const toast = document.getElementById('toast'); document.getElementById('toastText').innerText = text; toast.classList.add('show'); setTimeout(() => { toast.classList.remove('show'); }, 3000); }
